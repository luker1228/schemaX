<Takeaway
  num="CMP-10"
  label="总结"
  line1="基础组件会叫名"
  line2="区块级去进阶"
  note="基础组件是可复用的小零件；Hero / Navbar / Sidebar 是整页区块，进阶篇单独建词。"
>
  <ul>
    <li>
      <strong>展示 / 容器：</strong>
      Card · Modal · Drawer · Table
    </li>
    <li>
      <strong>操作 / 输入：</strong>
      Button · Form · Input
    </li>
    <li>
      <strong>状态：</strong>
      Loading · Empty · Badge——提示词要写全
    </li>
    <li>
      <strong>不在本课：</strong>
      Hero · Navbar · Sidebar 等区块 → 进阶「页面区块」
    </li>
  </ul>
</Takeaway>
